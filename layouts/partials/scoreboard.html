Other collections never get the fields. */}} {{- $ericPost := and (eq .Section "posts") (eq .Params.author "eric-wisnewski") -}} {{- $sb := .Params.scoreboard -}} {{- if and $ericPost $sb $sb.home $sb.away (isset $sb "home_score") (isset $sb "away_score") -}} {{- $home := $sb.home -}} {{- $away := $sb.away -}} {{- $homeScore := int $sb.home_score -}} {{- $awayScore := int $sb.away_score -}} {{- if lt $homeScore 0 }}{{ $homeScore = 0 }}{{ end -}} {{- if lt $awayScore 0 }}{{ $awayScore = 0 }}{{ end -}} {{- if gt $homeScore 999 }}{{ $homeScore = 999 }}{{ end -}} {{- if gt $awayScore 999 }}{{ $awayScore = 999 }}{{ end -}} {{- $width := 2 -}} {{- if or (ge $homeScore 100) (ge $awayScore 100) }}{{ $width = 3 }}{{ end -}} {{- end -}}
