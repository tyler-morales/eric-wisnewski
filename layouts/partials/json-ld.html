{{/* Structured data for Google: WebSite (home, hasPart = main-nav sections), CollectionPage/WebPage (those section landings), BlogPosting, Person. Skip noindex / 404 pages. No SearchAction — there is no site search. hasPart does not force sitelinks; Google still chooses them. Unique titles and descriptions on the nav landings are what the snippets are built from. */}} {{- if or (eq .Params.robots "noindex") (eq .Kind "404") -}} {{- else if .IsHome -}} {{- $site := dict "@context" "https://schema.org" "@type" "WebSite" "name" .Site.Title "url" .Site.BaseURL "description" (.Site.Params.description | default "") -}} {{- $parts := partial "nav-section-pages.html" . -}} {{- if $parts -}}{{- $site = merge $site (dict "hasPart" $parts) -}}{{- end -}} {{- else if and (eq .Section "authors") .IsPage -}} {{- $person := dict "@context" "https://schema.org" "@type" "Person" "name" (.Params.name | default .Title) "url" .Permalink -}} {{- with .Params.bio -}}{{- $person = merge $person (dict "description" .) -}}{{- end -}} {{- with .Params.image -}}{{- $person = merge $person (dict "image" (. | absURL)) -}}{{- end -}} {{- else if and .IsPage (in (slice "posts" "gradys-tour" "da-breakdown-w-tad" "jeremy-on-tap") .Section) -}} {{- $desc := .Summary | plainify | htmlUnescape | strings.TrimSpace -}} {{- if not $desc -}}{{- $desc = .Site.Params.description | default "" -}}{{- end -}} {{- $image := "images/og-default.jpg" | absURL -}} {{- with .Params.image -}}{{- $image = . | absURL -}}{{- end -}} {{- $author := dict "@type" "Person" "name" .Site.Title -}} {{- $authorsLd := slice -}} {{- range (partial "post-author-pages.html" .) -}} {{- $authorsLd = $authorsLd | append (dict "@type" "Person" "name" (.Params.name | default .Title) "url" .Permalink) -}} {{- end -}} {{- if eq (len $authorsLd) 1 -}} {{- $author = index $authorsLd 0 -}} {{- else if gt (len $authorsLd) 1 -}} {{- $author = $authorsLd -}} {{- end -}} {{- $post := dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" .Title "description" $desc "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "url" .Permalink "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "image" $image "author" $author "publisher" (dict "@type" "Organization" "name" .Site.Title "url" .Site.BaseURL) -}} {{- else if in (slice "page" "section") .Kind -}} {{- range $part := (partial "nav-section-pages.html" .) -}} {{- if eq (printf "%s" $part.url) (printf "%s" $.Permalink) -}} {{- $page := merge $part (dict "@context" "https://schema.org" "isPartOf" (dict "@type" "WebSite" "name" $.Site.Title "url" $.Site.BaseURL) ) -}} {{- end -}} {{- end -}} {{- end -}}
