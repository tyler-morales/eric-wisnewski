{{/* Resolve /authors// from dict site + slug. Falls back to matching content/authors filename or slug if they drifted. */}} {{- $key := .slug -}} {{- $site := .site -}} {{- $found := false -}} {{- with $key -}} {{- $found = $site.GetPage (printf "/authors/%s" .) -}} {{- end -}} {{- if not $found -}} {{- with $site.GetPage "/authors" -}} {{- range .Pages -}} {{- if or (eq .Params.slug $key) (eq .File.ContentBaseName $key) -}} {{- $found = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $found -}}
